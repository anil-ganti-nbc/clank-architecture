# systemd oneshot unit — one "Smartwatch Clank experimental soak" cycle.
# Alternative to deploy/crontab.example; pick one, not both.
# Pair with smartwatch-clank-soak.timer.example for unattended cadence.
#
# Install (example):
#   sudo cp deploy/smartwatch-clank-soak.service.example /etc/systemd/system/smartwatch-clank-soak.service
#   sudo cp deploy/smartwatch-clank-soak.timer.example   /etc/systemd/system/smartwatch-clank-soak.timer
#   # Edit WorkingDirectory first.
#   sudo systemctl daemon-reload
#   sudo systemctl enable --now smartwatch-clank-soak.timer
#
# One unit for the whole soak — new OEM collectors register into the same
# run via config, not a new scheduled unit per OEM.

# User=deploy/Group=deploy and StandardOutput/Error=journal match this
# host's existing convention (see korean-tech-wire-soak.service,
# oem-radar-experimental-sitemap-soak.service) -- deploy owns the staging
# checkout and is in the docker group.

[Unit]
Description=Smartwatch Clank experimental soak (one cycle)
After=network-online.target docker.service
Wants=network-online.target

[Service]
Type=oneshot
User=deploy
Group=deploy
WorkingDirectory=/home/deploy/staging/smartwatch-clank
ExecStart=/home/deploy/staging/smartwatch-clank/deploy/run.sh
Nice=10
TimeoutStartSec=1800
StandardOutput=journal
StandardError=journal
NoNewPrivileges=true
PrivateTmp=true

[Install]
WantedBy=multi-user.target

# systemd timer — matches the established 2-hour soak cadence
# (spec: avoid unnecessary changes to cadence; inspect existing fleet
# timers before picking the exact minute offset to avoid collisions).
#
# Fixed-clock OnCalendar (not OnUnitActiveSec) so the schedule is directly
# auditable against `systemctl list-timers` and the fleet's crontab, the
# same way every other Clank's timer/cron entry on this host is. Minute
# :12 at even hours was picked by inspecting the live fleet at deploy time
# (`systemctl list-timers --all` + `crontab -u deploy -l`): existing
# offsets in use were :05, :10, :15, :18, :20, :36, :40, and the existing
# smartwatch-clank PRODUCTION cron itself runs at :50 past ODD hours
# (`50 1-23/2 * * *`) -- this timer deliberately runs at even hours so the
# two never fire in the same wall-clock minute.
#
# Overlap is prevented by:
#   1) RandomizedDelaySec
#   2) the store's own RunLock (<database>.lock, single-instance run lock)
#      -- shared with the production cron path against the same database,
#      so even a coincidental overlap just makes one cycle wait, not race.

[Unit]
Description=Smartwatch Clank experimental soak, every 2 hours

[Timer]
OnCalendar=*-*-* 00/2:12:00
RandomizedDelaySec=90
Persistent=true
Unit=smartwatch-clank-soak.service

[Install]
WantedBy=timers.target
